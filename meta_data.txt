import java.util.*;

public class MergeSort {
    
    public static int[] merge(int[] a, int[] b) {
        // a and b are two sorted arrays
        // merge function will return a sorted array after merging a and b
        int[] result = new int[a.length + b.length];
        int i = 0, j = 0, k = 0;
        
        // Compare elements from both arrays and add the smaller one to result
        while (i < a.length && j < b.length) {
            if (a[i] <= b[j]) {
                result[k++] = a[i++];
            } else {
                result[k++] = b[j++];
            }
        }
        
        // Add remaining elements from a (if any)
        while (i < a.length) {
            result[k++] = a[i++];
        }
        
        // Add remaining elements from b (if any)
        while (j < b.length) {
            result[k++] = b[j++];
        }
        
        return result;
    }
    
    public static int[] mergeSort(int[] arr) {
        if (arr.length <= 1) {
            return arr;
        } else {
            int mid = arr.length / 2;
            int[] a1 = mergeSort(Arrays.copyOfRange(arr, 0, mid));  // left half
            int[] a2 = mergeSort(Arrays.copyOfRange(arr, mid, arr.length));  // right half
            return merge(a1, a2);
        }
    }
    
    // Helper method to print array
    public static void printArray(int[] arr) {
        for (int num : arr) {
            System.out.print(num + " ");
        }
        System.out.println();
    }
    
    // Main method to test the implementation
    public static void main(String[] args) {
        int[] arr = {64, 34, 25, 12, 22, 11, 90};
        System.out.print("Original array: ");
        printArray(arr);
        
        int[] sortedArr = mergeSort(arr);
        System.out.print("Sorted array: ");
        printArray(sortedArr);
    }
}
